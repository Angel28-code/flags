filteredFlags = {
  if (category === "2-3 Colors") return flags.filter(d => d.num_colors <= 3);
  if (category === "4+ Colors") return flags.filter(d => d.num_colors >= 4);
  if (category === "Low Complexity") return flags.filter(d => d.complexity === "Low");
  if (category === "High Complexity") return flags.filter(d => d.complexity === "High");
  if (category === "Symmetric") return flags.filter(d => d.symmetry !== "Asymmetric");
  if (category === "Asymmetric") return flags.filter(d => d.symmetry === "Asymmetric");
  if (category === "High Distinctiveness") return flags.filter(d => d.distinctiveness === "High");
  return flags;
}
